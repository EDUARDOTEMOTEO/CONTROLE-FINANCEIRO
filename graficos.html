import pandas as pd import plotly.express as px # Lê a tabela de lançamentos df = pd.read_html("tabela.html")[0] # Limpa e converte os dados df["Valor (R$)"] = pd.to_numeric(df["Valor (R$)"], errors="coerce") df["Despesas"] = pd.to_numeric(df["Despesas"], errors="coerce") df["Data"] = pd.to_datetime(df["Data"], errors="coerce") df = df.dropna(subset=["Data", "Tipo", "Valor (R$)", "Despesas", "Categoria"]) # Filtra entradas e saídas df_entrada = df[df["Tipo"].str.lower() == "entrada"] df_saida = df[df["Tipo"].str.lower().isin(["saída", "saida"])] # Gráfico de barras por categoria de despesa bar = px.bar(df_saida.groupby("Categoria", as_index=False).sum(), x="Categoria", y="Despesas", title="💸 Despesas por Categoria") bar.write_html("bar_categoria.html") # Gráfico de linha temporal de lançamentos linha = px.line(df.groupby(["Data", "Tipo"], as_index=False).sum(), x="Data", y="Valor (R$)", color="Tipo", title="📆 Fluxo Financeiro ao Longo do Tempo", markers=True) linha.write_html("linha_temporal.html") # Gráfico de pizza: entradas vs saídas pizza = px.pie(df, names="Tipo", values="Valor (R$)", title="🧾 Entradas vs Saídas") pizza.write_html("pizza_movimento.html")
